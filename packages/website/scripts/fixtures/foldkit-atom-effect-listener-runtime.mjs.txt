import assert from 'node:assert/strict'
import { readFileSync } from 'node:fs'
import { pathToFileURL } from 'node:url'
import fc from 'fast-check'
import { Window } from 'happy-dom'
const window=new Window()
for(const name of ['window','document','navigator','HTMLElement','Node','Event','MouseEvent'])Object.defineProperty(globalThis,name,{value:name==='window'?window:window[name],configurable:true,writable:true})
const {Context,Layer,Effect}=await import('effect')
const {Atom,AtomRegistry}=await import('effect/reactivity')
const corpus=JSON.parse(readFileSync('foldkit-atom-effect-listener-runtime.json','utf8'))
let stats;const active=new Map();const add=window.addEventListener.bind(window);const remove=window.removeEventListener.bind(window)
window.addEventListener=(type,listener,options)=>{
 if(type!=='mouseup')return add(type,listener,options)
 const captured=stats
 const wrapped=event=>{captured.delivered++;if(typeof listener==='function')listener.call(window,event);else listener.handleEvent(event)}
 assert.ok(!active.has(listener),'listener identity is not double registered');active.set(listener,{wrapped,options})
 captured.adds++;return add(type,wrapped,options)
}
window.removeEventListener=(type,listener,options)=>{
 if(type!=='mouseup')return remove(type,listener,options)
 const entry=active.get(listener);assert.ok(entry,'removed listener was registered')
 const capture=x=>typeof x==='boolean'?x:x?.capture??false
 assert.equal(capture(options),capture(entry.options));active.delete(listener);stats.removes++;return remove(type,entry.wrapped,options)
}
const variant=process.argv[2]??'native';let sequence=0,models=0,events=0,mounts=0,writes=0
const tick=()=>new Promise(resolve=>setTimeout(resolve,0))
async function run(side,model){
 assert.equal(active.size,0)
 stats={adds:0,removes:0,delivered:0,writes:[]}
 const local=stats;const key=Context.Service('Effect.Api.'+sequence)
 const service={getUser:()=>Effect.succeed({name:'unused'}),createTodo:text=>Effect.sync(()=>{local.writes.push(text);writes++})}
 globalThis.Api=Object.assign(key,{Default:Layer.succeed(key,service)})
 globalThis.runtime=Atom.runtime(Api.Default)
 const path=process.cwd()+'/runtime-output/atomCompareAtomEffect-'+side+'/atomCompareAtomEffect.js'
 const mod=await import(pathToFileURL(path).href+'?model='+sequence++)
 assert.equal(local.adds,0,'module construction does not register a listener');assert.equal(local.writes.length,0)
 const reg=AtomRegistry.make({defaultIdleTTL:0,scheduleTask:f=>{f();return()=>{}}});const stops=[];const observed=[]
 const snapshot=(op,phase)=>({op,phase,active:active.size,adds:local.adds,removes:local.removes,delivered:local.delivered,writes:[...local.writes]})
 try{
  for(let i=0;i<model.ops.length;i++){
   const op=model.ops[i]
   if(op==='mount'){stops.push(reg.mount(mod.mouseUpAtom));mounts++}
   else if(op==='unmount'){stops.pop()?.()}
   else if(op==='event'){window.dispatchEvent(new window.MouseEvent('mouseup'));events++}
   else if(op==='write')reg.set(mod.createTodoAtom,model.text+'-'+i)
   if(stops.length>0)assert.equal(reg.get(mod.mouseUpAtom),false)
   observed.push(snapshot(op,'synchronous'));await tick();observed.push(snapshot(op,'settled'))
  }
 }finally{
  while(stops.length)stops.pop()()
  reg.dispose();observed.push(snapshot('dispose','synchronous'));await tick();observed.push(snapshot('dispose','settled'))
 }
 assert.equal(active.size,0,'registry disposal removes every browser listener')
 assert.equal(local.adds,local.removes)
 return observed
}
async function compare(model){
 const before=JSON.stringify(model);const original=await run('original',model);const native=await run(variant==='mutant'?'mutants':'native',model)
 assert.deepEqual(native,original);assert.equal(JSON.stringify(model),before);models++
}
try{
 for(const model of corpus.fixed)await compare(model)
 await fc.assert(fc.asyncProperty(fc.record({ops:fc.array(fc.constantFrom('mount','unmount','event','write'),{maxLength:16}),text:fc.string({maxLength:16})}),compare),{seed:corpus.seed,numRuns:corpus.numRuns})
 console.log('ATOM_EFFECT_LISTENER_RUNTIME '+JSON.stringify({seed:corpus.seed,models,generatedModels:100,fixedModels:2,events,mounts,writes,variant,realWindowAndAtomRegistry:true,coldConstruction:true,synchronousAndSettledTracesIdentical:true,inputUnchanged:true}))
}finally{window.close()}
