import assert from 'node:assert/strict'
import { pathToFileURL } from 'node:url'
import { readFileSync } from 'node:fs'
import fc from 'fast-check'
import { Window } from 'happy-dom'
import { Context, Effect, Layer } from 'effect'
import { Atom, AtomRegistry } from 'effect/reactivity'
const corpus = JSON.parse(readFileSync('foldkit-atom-effect-mutation-runtime.json', 'utf8'))
const window = new Window()
Object.defineProperty(globalThis, 'window', { value: window, configurable: true })
Object.defineProperty(globalThis, 'document', { value: window.document, configurable: true })
let sequence = 0
const tick = () => new Promise(resolve => setTimeout(resolve, 0))
async function run(side, texts) {
  const calls = []
  const runs = []
  const Api = Context.Service('Mutation.Api.' + sequence)
  globalThis.Api = Object.assign(Api, {
    Default: Layer.succeed(Api, {
      createTodo(text) {
        calls.push(text)
        return Effect.sync(() => { runs.push(text) })
      },
    }),
  })
  globalThis.runtime = Atom.runtime(Api.Default)
  const registry = AtomRegistry.make({ defaultIdleTTL: 0, scheduleTask: callback => { callback(); return () => {} } })
  const file = process.cwd() + '/runtime-output/atomCompareAtomEffect-' + side + '/atomCompareAtomEffect.js'
  const module = await import(pathToFileURL(file).href + '?model=' + sequence++)
  assert.deepEqual(calls, [], 'constructing the atom stays cold')
  const initial = registry.get(module.createTodoAtom)
  assert.deepEqual(calls, [], 'reading its initial result stays cold')
  const states = [initial._tag]
  try {
    for (const text of texts) {
      const count = calls.length
      registry.set(module.createTodoAtom, text)
      for (let i = 0; calls.length === count && i < 30; i++) await tick()
      assert.equal(calls.length, count + 1, 'one service call per write')
      assert.equal(calls.at(-1), text)
      for (let i = 0; runs.length < calls.length && i < 30; i++) await tick()
      assert.deepEqual(runs, calls)
      states.push(registry.get(module.createTodoAtom)._tag)
    }
    return { calls, runs, states }
  } finally {
    registry.dispose()
    await tick()
  }
}
let models = 0
let writes = 0
try {
  const compare = async texts => {
    const frozen = JSON.stringify(texts)
    const original = await run('original', texts)
    const native = await run(process.argv[2] === 'mutant' ? 'mutants' : 'native', texts)
    assert.deepEqual(native, original)
    assert.equal(JSON.stringify(texts), frozen, 'input text array is unchanged')
    assert.equal(native.calls.length, texts.length)
    assert.deepEqual(native.calls, texts)
    models++
    writes += texts.length
  }
  for (const texts of corpus.fixed) await compare(texts)
  await fc.assert(fc.asyncProperty(
    fc.array(fc.oneof(fc.string({ maxLength: 24 }), fc.constantFrom('', 'repeat', 'café🔥', '<&☃')), { minLength: 1, maxLength: 8 }),
    compare,
  ), { seed: corpus.seed, numRuns: corpus.numRuns, endOnFailure: true })
  console.log('ATOM_EFFECT_MUTATION_RUNTIME ' + JSON.stringify({
    seed: corpus.seed, models, generatedModels: corpus.numRuns,
    fixedModels: corpus.fixed.length, writes, variant: 'native',
    realServiceAndAtomRegistry: true, coldConstruction: true,
    exactTextAndOperationCount: true, resultTagsIdentical: true,
    inputUnchanged: true,
  }))
} finally { window.close() }
